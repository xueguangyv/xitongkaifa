import subprocess

def test_normal_name():
    result = subprocess.run(
        ["sdt-greet", "--name", "Alice"],
        capture_output=True,
        text=True
    )
    assert result.returncode == 0
    assert "Hello, Alice!" in result.stdout

def test_blank_name():
    result = subprocess.run(
        ["sdt-greet", "--name", ""],
        capture_output=True,
        text=True
    )
    assert result.returncode == 0
    assert "Hello, !" in result.stdout
EOF

#!/bin/bash
curl -fsS http://127.0.0.1:8000/packages.json | jq '
.[]
| select(.status == "active" and .downloads >= 100)
' | jq -s 'sort_by(-.downloads, .name)' > filtered.json

# Active Packages Summary

| name | version | downloads |
|------|---------|-----------|
MD

rm -f filtered.json
EOF
